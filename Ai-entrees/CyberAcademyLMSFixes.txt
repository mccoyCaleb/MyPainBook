```js
document.querySelectorAll('.accordion-toggle').forEach((button) => {
  const item = button.closest('.accordion-item');
  const panel = item ? item.querySelector('.accordion-panel') : null;
  if (!panel) return;
  panel.hidden = !item.classList.contains('open');
  button.addEventListener('click', () => {
    const shouldOpen = !item.classList.contains('open');
    item.classList.toggle('open', shouldOpen);
    button.setAttribute('aria-expanded', String(shouldOpen));
    panel.hidden = !shouldOpen;
  });
});
```

This restores the original accordion behavior so lesson sections expand and collapse like the original course pages. Without this, users could see the content but the interactive slide-down sections would not behave correctly.

```js
document.querySelectorAll('.quiz-card').forEach((card) => {
  const correctValue = card.dataset.correct;
  const buttons = card.querySelectorAll('.option-button');
  const answerBox = card.querySelector('.answer-box');

  buttons.forEach((button) => {
    button.addEventListener('click', () => {
      const selected = button.dataset.answer;
      buttons.forEach((btn) => {
        btn.disabled = true;
        btn.classList.remove('selected', 'correct', 'incorrect');
        if (btn.dataset.answer === correctValue) {
          btn.classList.add('correct');
        }
      });

      button.classList.add('selected');
      if (selected === correctValue) {
        button.classList.add('correct');
      } else {
        button.classList.add('incorrect');
      }

      if (answerBox) {
        answerBox.hidden = false;
      }
    });
  });
});
```

This brings back the original question/reveal behavior. The buttons act like real multiple-choice interactions instead of just static text, and the answer box opens after selection.
